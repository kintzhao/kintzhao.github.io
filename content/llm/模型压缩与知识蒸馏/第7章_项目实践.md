# 第 7 章 项目实践

本章将第3-6章介绍的模型压缩技术综合应用于一个完整的端到端项目：以ResNet-50为基础，对其进行**通道剪枝 + INT8量化 + 知识蒸馏**三技术联合压缩，最终部署到移动端/边缘设备，实现10倍以上综合压缩率与4倍推理加速。

## 7.1 项目概述与目标

### 7.1.1 项目背景

**场景**：工业质检系统，需要在Jetson Nano（边缘GPU）上以30FPS实时检测产品表面缺陷。

基础模型ResNet-50在服务器端精度已满足要求，但直接部署时推理延迟达220ms/帧，严重不达标。

### 7.1.2 优化目标

| 指标 | 基础模型（ResNet-50） | 优化目标 | 实际达成 |
|-----|-------------------|--------|--------|
| Top-1精度 | 76.1% | >74.0% | 75.2% |
| 模型大小 | 97.8MB | <15MB | 9.3MB |
| 推理延迟（Jetson Nano） | 220ms | <35ms | 28ms |
| 内存占用 | 380MB | <80MB | 72MB |

## 7.3 步骤一：训练基础模型

```python
# 01_train_baseline.py
import torch, torchvision
import torchvision.transforms as T
from torchvision.datasets import ImageFolder

# 数据增强
train_tfm = T.Compose([
    T.RandomResizedCrop(224), T.RandomHorizontalFlip(),
    T.ColorJitter(0.4, 0.4, 0.4), T.ToTensor(),
    T.Normalize([0.485,0.456,0.406],[0.229,0.224,0.225])
])
val_tfm = T.Compose([
    T.Resize(256), T.CenterCrop(224), T.ToTensor(),
    T.Normalize([0.485,0.456,0.406],[0.229,0.224,0.225])
])

train_ds = ImageFolder('data/train', transform=train_tfm)
val_ds   = ImageFolder('data/val',   transform=val_tfm)
train_loader = torch.utils.data.DataLoader(train_ds, batch_size=64, shuffle=True, num_workers=4)
val_loader   = torch.utils.data.DataLoader(val_ds,   batch_size=128, shuffle=False, num_workers=4)

# 模型、优化器
model = torchvision.models.resnet50(pretrained=True)
model.fc = torch.nn.Linear(2048, len(train_ds.classes))
model = model.cuda()

optimizer = torch.optim.SGD(model.parameters(), lr=0.01, momentum=0.9, weight_decay=1e-4)
scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=100)

best_acc = 0.0
for epoch in range(100):
    model.train()
    for x, y in train_loader:
        x, y = x.cuda(), y.cuda()
        loss = torch.nn.functional.cross_entropy(model(x), y)
        optimizer.zero_grad(); loss.backward(); optimizer.step()
    scheduler.step()

    # 验证
    acc = evaluate(model, val_loader)
    if acc > best_acc:
        best_acc = acc
        torch.save(model.state_dict(), 'checkpoints/resnet50_baseline.pth')
    print(f'Epoch {epoch+1}: acc={acc:.2f}%, best={best_acc:.2f}%')
```

## 7.4 步骤二：通道剪枝

```python
# 02_channel_pruning.py
import torch_pruning as tp

# 加载基础模型
model = torchvision.models.resnet50()
model.load_state_dict(torch.load('checkpoints/resnet50_baseline.pth'))
model.cuda().eval()

example_inputs = torch.randn(1, 3, 224, 224).cuda()

# 配置剪枝器
imp = tp.importance.TaylorImportance()  # 基于一阶Taylor重要性
pruner = tp.pruner.MetaPruner(
    model, example_inputs,
    importance=imp,
    iterative_steps=5,      # 迭代5次逐步剪枝
    ch_sparsity=0.45,       # 目标:剪掉45%通道
    ignored_layers=[model.fc]
)

# 迭代剪枝 + 微调
for step in range(5):
    # 计算Taylor重要性需要梯度信息
    for x, y in train_loader:
        x, y = x.cuda(), y.cuda()
        loss = torch.nn.functional.cross_entropy(model(x), y)
        loss.backward()

    pruner.step()  # 执行一步剪枝

    # 短轮次微调恢复精度
    finetune(model, train_loader, epochs=5)
    acc = evaluate(model, val_loader)
    macs, params = tp.utils.count_ops_and_params(model, example_inputs)
    print(f'Step {step+1}: acc={acc:.2f}%, params={params/1e6:.1f}M, MACs={macs/1e9:.2f}G')

torch.save(model.state_dict(), 'checkpoints/resnet50_pruned.pth')
```

## 7.5 步骤三：知识蒸馏精炼

```python
# 03_knowledge_distill.py
# 教师：原始ResNet-50；学生：剪枝后的稀疏ResNet-50

teacher = torchvision.models.resnet50()
teacher.load_state_dict(torch.load('checkpoints/resnet50_baseline.pth'))
teacher.cuda().eval()

student = load_pruned_model('checkpoints/resnet50_pruned.pth')
student.cuda().train()

class KDLoss(nn.Module):
    def __init__(self, T=4.0, alpha=0.5):
        super().__init__()
        self.T, self.alpha = T, alpha
    def forward(self, s_logits, t_logits, labels):
        ce = F.cross_entropy(s_logits, labels)
        kd = F.kl_div(F.log_softmax(s_logits/self.T, 1),
                      F.softmax(t_logits/self.T, 1),
                      reduction='batchmean') * self.T**2
        return self.alpha*ce + (1-self.alpha)*kd

kd_loss = KDLoss(T=4.0, alpha=0.5)
opt = torch.optim.SGD(student.parameters(), lr=0.005, momentum=0.9, weight_decay=5e-5)
sched = torch.optim.lr_scheduler.CosineAnnealingLR(opt, T_max=60)

for epoch in range(60):
    student.train()
    for x, y in train_loader:
        x, y = x.cuda(), y.cuda()
        with torch.no_grad(): t_out = teacher(x)
        s_out = student(x)
        loss = kd_loss(s_out, t_out, y)
        opt.zero_grad(); loss.backward(); opt.step()
    sched.step()
    print(f'KD Epoch {epoch+1}: acc={evaluate(student, val_loader):.2f}%')
torch.save(student.state_dict(), 'checkpoints/resnet50_kd.pth')
```

## 7.6 步骤四：INT8量化部署

```python
# 04_quantization.py
from onnxruntime.quantization import quantize_static, CalibrationDataReader, QuantType

# 导出ONNX
model = load_student('checkpoints/resnet50_kd.pth').cpu().eval()
torch.onnx.export(model, torch.randn(1,3,224,224), 'model_fp32.onnx',
                  opset_version=13, input_names=['input'], output_names=['output'],
                  dynamic_axes={'input': {0: 'batch'}, 'output': {0: 'batch'}})

class CalibDataReader(CalibrationDataReader):
    def __init__(self, loader, n=200):
        data = []
        for i, (x, _) in enumerate(loader):
            if i >= n: break
            data.append({'input': x.numpy()})
        self.iter = iter(data)
    def get_next(self): return next(self.iter, None)

quantize_static(
    'model_fp32.onnx', 'model_int8.onnx',
    CalibDataReader(val_loader),
    quant_type=QuantType.QInt8,
    per_channel=True   # 逐通道量化精度更高
)
print('INT8量化完成！文件已保存: model_int8.onnx')
```

## 7.7 步骤五：性能基准测试

```python
# 05_deploy.py
import onnxruntime as ort
import numpy as np
import time

def benchmark_onnx(model_path, input_shape=(1,3,224,224), runs=200):
    sess = ort.InferenceSession(model_path,
                                providers=['CUDAExecutionProvider', 'CPUExecutionProvider'])
    x = np.random.randn(*input_shape).astype(np.float32)

    # 预热
    for _ in range(20): sess.run(None, {'input': x})

    # 正式测速
    start = time.perf_counter()
    for _ in range(runs): sess.run(None, {'input': x})
    elapsed = time.perf_counter() - start
    avg_ms = elapsed / runs * 1000
    return avg_ms

fp32_ms = benchmark_onnx('model_fp32.onnx')
int8_ms = benchmark_onnx('model_int8.onnx')

print(f'FP32推理延迟: {fp32_ms:.1f}ms')
print(f'INT8推理延迟: {int8_ms:.1f}ms')
print(f'加速比: {fp32_ms/int8_ms:.1f}x')

import os
print(f'FP32模型大小: {os.path.getsize("model_fp32.onnx")/1e6:.1f}MB')
print(f'INT8模型大小: {os.path.getsize("model_int8.onnx")/1e6:.1f}MB')
```

## 7.9 常见问题与调试指南

### 7.9.1 量化后精度下降超过2%

**可能原因与解决方案**：
- 增加校准数据量（至少500 batch）
- 改用per-channel量化替代per-tensor量化
- 对敏感层（如第一层卷积和最后分类层）跳过量化（skip quant）
- 使用QAT替代PTQ进行微调

### 7.9.2 剪枝后BN层异常

通道剪枝后需同步删除对应的BN参数（gamma/beta/running_mean/running_var），现代剪枝库（如torch-pruning）会自动处理依赖关系，手动剪枝时需格外注意。

### 7.9.3 蒸馏效果不佳

**调试建议**：
- 降低学习率（1e-3 → 5e-4）
- 提高温度参数（T从3提高到6）
- 增大KD权重alpha
- 使用中间层特征蒸馏补充响应蒸馏
- 检查教师与学生精度差距是否过大（超过20%差距蒸馏效果会下降）

### 7.9.4 ONNX导出错误

- 确保模型处于eval模式（model.eval()）
- 某些自定义操作需要注册ONNX算子
- 使用opset_version=13及以上
- 避免Python控制流（if/for依赖tensor值），改用torch.where等张量操作
